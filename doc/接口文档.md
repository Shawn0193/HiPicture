# 数据库设计
```sql
USE my_picture;

CREATE TABLE `picture` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `cover_pictures` varchar(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '封面图，$拼接',
  `title` varchar(50)  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '标题',
  `introduce` varchar(200)  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '介绍',
  `user_name` varchar(50)  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '作者',
  `category` varchar(50)  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '类型',
  `width` int unsigned NOT NULL COMMENT '宽度',
  `height` int unsigned NOT NULL COMMENT '高度',
  `size` int unsigned NOT NULL COMMENT '大小',
  `create_time` int unsigned NOT NULL,
  `update_time` int unsigned NOT NULL,
  `is_deleted` tinyint unsigned NOT NULL DEFAULT '0',
  PRIMARY KEY (`picture_id`),
  KEY `idx_user_name` (`user_name`),
  KEY `idx_category` (`category`)
)ENGINE=InnoDb DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='图片表'
```
